# 服务器配置指令

## MySQL的配置

```bash
apt-get install mysql-server
mysqld --initialize
mysqladmin --version
mysqladmin -u root password "rootpassword";
mysql -u root -p # 连接服务器
CREATE USER 'ncnet'@'localhost' IDENTIFIED BY 'dbuserpassword';
CREATE DATABASE ncnetdb;
GRANT ALL PRIVILEGES ON ncnetdb.* TO 'ncnet'@'localhost';
FLUSH PRIVILEGES;
SHOW GRANTS FOR 'ncnet'@'localhost'; # 查看用户权限
```

## 配置python

```bash
mkdir /www
mkdir /www/server
cd /www/server
git clone https://github.com/NCBCNet/NCBCNet.git
cd NCBCNet
sudo apt-get install python3 python3-venv -y
python3 -m venv env
apt-get install pkg-config
sudo apt install default-libmysqlclient-dev
source env/bin/activate
pip install -r requirements.txt
python manage.py collectstatic
python manage.py makemigrations
python manage.py migrate
python manage.py createsuperuser
```



## 配置certbot

```bash
pip install certbot
pip install certbot-dns-cloudflare
certbot --version
sudo mkdir -p /etc/letsencrypt
sudo vim /etc/letsencrypt/cloudflare.ini
# 输入文件
dns_cloudflare_api_token = YOUR_CLOUDFLARE_API_TOKEN

sudo chmod 600 /etc/letsencrypt/cloudflare.ini
certbot certonly --dns-cloudflare --dns-cloudflare-credentials /etc/letsencrypt/cloudflare.ini -d ncnetstudent.top -d www.ncnetstudent.top
certbot renew --dry-run
systemctl status cron
```

## 配置systemd

```bash
ln -s systemd/daphne.service /etc/systemd/system
```

## 配置nginx

```
vim /etc/nginx/sites-available/ncnetstudent.top
# 复制内容
ln -s /etc/nginx/sites-available/ncnetstudent.top /etc/nginx/sites-enabled
nginx -t
systemctl reload nginx
```
