# 服务器配置指令

## PostgreSQL 的配置

```bash
apt-get install postgresql postgresql-contrib
systemctl enable --now postgresql
sudo -u postgres psql  # 连接服务器
CREATE USER ncnet WITH PASSWORD 'dbuserpassword';
CREATE DATABASE ncnetdb OWNER ncnet;
GRANT ALL PRIVILEGES ON DATABASE ncnetdb TO ncnet;
\du   # 查看用户
\l    # 查看数据库
\q    # 退出 psql
```

## 配置python

```bash
mkdir /www
mkdir /www/server
cd /www/server
git clone https://github.com/NCBCNet/NCBCNet.git
cd NCBCNet
sudo apt-get install python3 python3-venv -y
python3 -m venv env
apt-get install pkg-config
sudo apt install postgresql-client   # psql/pg_dump（备份脚本与连接所需）；psycopg[binary] 为预编译 wheel，无需系统库
source env/bin/activate
pip install -r requirements.txt
python manage.py collectstatic
python manage.py makemigrations
python manage.py migrate
python manage.py createsuperuser
deactivate # 退出虚拟环境
```



## 配置certbot

```bash
pip install certbot
pip install certbot-dns-cloudflare
certbot --version
sudo mkdir -p /etc/letsencrypt
sudo vim /etc/letsencrypt/cloudflare.ini
# 输入文件
dns_cloudflare_api_token = YOUR_CLOUDFLARE_API_TOKEN

sudo chmod 600 /etc/letsencrypt/cloudflare.ini
certbot certonly --dns-cloudflare --dns-cloudflare-credentials /etc/letsencrypt/cloudflare.ini -d ncnetstudent.top -d www.ncnetstudent.top
certbot renew --dry-run
systemctl status cron
```

## 配置systemd

```bash
ln -s systemd/daphne.service /etc/systemd/system
```

## 配置nginx

```bash
vim /etc/nginx/sites-available/ncnetstudent.top
# 复制内容
ln -s /etc/nginx/sites-available/ncnetstudent.top /etc/nginx/sites-enabled
nginx -t
systemctl reload nginx
```
